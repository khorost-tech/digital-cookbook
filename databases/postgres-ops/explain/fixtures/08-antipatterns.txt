# Прогон 08-antipatterns.sql (PostgreSQL 18.4, 2026-10-04; перед ним 00–07 по порядку)
# cmd: ./run.sh sql/08-antipatterns.sql
# Диагноз по плану: Seq Scan/Filter там, где ждали Index Cond.
# «Как надо» обязано возвращать те же строки — проверка обеими разностями EXCEPT ALL.

=== 1. функция по колонке ===
# СЛОМАНО: lower(name) = 'ivan'
 Seq Scan on customers  (cost=0.00..1211.00 rows=250 width=42) (actual ... rows=8384.00 ...)
   Filter: (lower(name) = 'ivan'::text)
   Rows Removed by Filter: 41616
 Execution Time: 5.103 ms
# НЕ ЗАМЕНА: name = 'ivan' — индекс работает, но запрос другой
 Bitmap Heap Scan on customers  (... rows=2793.00 ...)
   Recheck Cond: (name = 'ivan'::text)
   ->  Bitmap Index Scan on idx_customers_name
         Index Cond: (name = 'ivan'::text)
 Execution Time: 0.334 ms
# проверка эквивалентности:
 lost_rows | extra_rows
-----------+------------
      5591 |          0
# Быстрее, но потеряны 5591 строка 'Ivan' из 8384. Это не оптимизация, а другой результат.
# КАК НАДО: тот же lower(name) = 'ivan' + CREATE INDEX ON customers (lower(name))
 Index Scan using idx_customers_lower_name on customers  (... rows=8384.00 ...)
   Index Cond: (lower(name) = 'ivan'::text)
 Execution Time: 0.726 ms
# 5.1 мс → 0.73 мс (~7×), те же 8384 строки. Expression-индекс возвращает предикат в Index Cond.

=== 2. вычисление/каст по колонке ===
# СЛОМАНО: created_at::date = current_date
 Aggregate
   ->  Seq Scan on orders  (cost=0.00..51959.00 rows=10000 ...) (actual ... rows=1191.00 ...)
         Filter: ((created_at)::date = CURRENT_DATE)
         Rows Removed by Filter: 1998809
 Execution Time: 96.051 ms
# Каст по колонке → нет Index Cond, читается вся таблица (2 млн строк, отброшено 1.99M).
# КАК НАДО: полуинтервал по самой колонке
 Aggregate
   ->  Index Only Scan using idx_orders_created on orders  (cost=0.43..38.51 rows=1179 ...)
         Index Cond: ((created_at >= CURRENT_DATE) AND (created_at < (CURRENT_DATE + 1)))
 Execution Time: 0.138 ms
# проверка эквивалентности (по id):
 lost_rows | extra_rows
-----------+------------
         0 |          0
# 96 мс → 0.14 мс (~700×), строки те же. Тот же индекс, но теперь Index Cond, а не Filter.

=== 3. OR: миф и реальность ===
# НЕ антипаттерн: оба операнда селективны и под индексами → BitmapOr
 Bitmap Heap Scan on orders  (... rows=104.00 ...)
   Recheck Cond: ((customer_id = 12345) OR (created_at >= (now() - '00:30:00'::interval)))
   ->  BitmapOr
         ->  Bitmap Index Scan on idx_orders_customer   (Index Cond: customer_id = 12345)
         ->  Bitmap Index Scan on idx_orders_created    (Index Cond: created_at >= ...)
 Execution Time: 0.046 ms
# АНТИПАТТЕРН: один операнд без индекса (status) → весь OR уходит в Seq Scan
 Seq Scan on orders  (... rows=250349.00 ...)
   Filter: ((customer_id = 12345) OR (status = 'cancelled'::text))
 Execution Time: 61.980 ms
# Вывод: OR не зло. Зло — неиндексируемая ветвь: достаточно одной, чтобы убить весь OR.

=== 4. LIKE с ведущим % ===
# СЛОМАНО: name LIKE '%van%'
 Seq Scan on customers  (... rows=8384.00 ...)
   Filter: (name ~~ '%van%'::text)
 Execution Time: 1.626 ms
# НЕ ЗАМЕНА: name LIKE 'Iva%' с индексом text_pattern_ops — префиксный поиск, запрос другой
 Bitmap Heap Scan on customers  (... rows=5591.00 ...)
   ->  Bitmap Index Scan on idx_customers_name_pat
         Index Cond: ((name ~>=~ 'Iva'::text) AND (name ~<~ 'Ivb'::text))
 Execution Time: 0.558 ms
# проверка эквивалентности:
 lost_rows | extra_rows
-----------+------------
      2793 |          0
# Потеряны 2793 строки 'ivan'. Префикс годится, только если по задаче нужен именно префикс.
# КАК НАДО: тот же LIKE '%van%' + CREATE INDEX ... USING gin (name gin_trgm_ops)
 Bitmap Heap Scan on customers  (... rows=8384.00 ...)
   Recheck Cond: (name ~~ '%van%'::text)
   ->  Bitmap Index Scan on idx_customers_name_trgm
         Index Cond: (name ~~ '%van%'::text)
 Execution Time: 0.878 ms
# 1.6 мс → 0.88 мс (~2×) на 50 тыс. строк и 17% совпадений; выигрыш растёт с размером таблицы
# и селективностью подстроки. Главное — план идёт через индекс, а строки те же.
