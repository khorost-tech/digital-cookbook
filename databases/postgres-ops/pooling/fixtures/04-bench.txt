# Прогон bench.sh (PostgreSQL 18.4, PgBouncer 1.25.2 / pgcat 1.2.0 / Odyssey 1.5.1, 2026-07-22)
# Где пулер реально помогает. Read-only (-S). max_connections=100.

=== A. Перегрузка: 150 клиентов при max_connections=100 ===
напрямую к postgres   FATAL: sorry, too many clients already   ← база ОТКАЗЫВАЕТ
через PgBouncer       tps = 13550
через pgcat           tps = 21999
через Odyssey         tps =  9195
# Главный аргумент за пулер: 150 клиентов не помещаются в 100 соединений. Напрямую
# лишние получают отказ. Пулер мультиплексирует их на 5 server-соединений — все работают.

=== B. Переподключение (-C): новое соединение на каждую транзакцию ===
напрямую к postgres   tps = 2144   (avg connection 3.68 ms)   ← форк процесса на каждую
через PgBouncer       tps = 4043   (avg connection 1.93 ms)
через pgcat           tps = 4759   (avg connection 1.65 ms)
через Odyssey         tps = 4824   (avg connection 1.13 ms)
# Установка соединения к postgres = форк backend-процесса (дорого). Пулер держит
# server-соединения готовыми и переиспользует — переподключение вдвое дешевле и быстрее.

# Различия между пулерами (pgcat лидирует в A, Odyssey в B) — реальная разница реализаций
# (pgcat и Odyssey на Rust/многопоточные, PgBouncer однопоточный), но все три решают
# главную задачу: держат базу под лавиной клиентских соединений.
# Замечание: устойчивую нагрузку тут не мерили — при pool_size=5 малый пул был бы
# искусственным горлышком. Пул подбирают под число ядер БД, это отдельная настройка.
