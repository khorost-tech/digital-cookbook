# Прогон 01-connection-cost.sql + conn-demo.sh (PostgreSQL 18.4, 2026-07-22)
# Соединение к PG = отдельный процесс-бэкенд. Пулер мультиплексирует клиентов на немногие.

=== соединение = процесс ===
 pid |  backend_type  | state  | application_name
-----+----------------+--------+------------------
  79 | client backend | idle   | psql
 ...
 max_connections = 100        -- жёсткий предел прямых соединений
 shared_buffers  = 256MB
 work_mem        = 4MB        -- множится на число бэкендов и сложных узлов

=== мультиплексирование: 50 клиентов на каждый endpoint ===
напрямую (без пулера)          server-бэкендов при 50 клиентах: 50
через PgBouncer (pool=5)        server-бэкендов при 50 клиентах: 5
через pgcat (pool=5)            server-бэкендов при 50 клиентах: 10
через Odyssey (pool=5)          server-бэкендов при 50 клиентах: 15

# Напрямую: 50 клиентов = 50 процессов на сервере (по процессу на соединение).
# Через любой пулер: 50 клиентов обслуживаются горсткой реальных соединений.
# PgBouncer строго держит default_pool_size=5. pgcat и Odyssey под нагрузкой держат
# чуть больше (10 и 15) — у каждого своя политика роста/резерва пула, но все на порядок
# ниже 50. Именно это спасает max_connections на сотнях приложений.
