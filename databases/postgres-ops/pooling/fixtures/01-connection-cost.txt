# Прогон 01-connection-cost.sql + conn-demo.sh (PostgreSQL 18.4, 2026-07-22)
# Соединение к PG = отдельный процесс-бэкенд. Пулер мультиплексирует клиентов на немногие.

=== соединение = процесс ===
 pid |  backend_type  | state  | application_name
-----+----------------+--------+------------------
  79 | client backend | idle   | psql
 ...
 max_connections = 100        -- жёсткий предел прямых соединений
 shared_buffers  = 256MB
 work_mem        = 4MB        -- множится на число бэкендов и сложных узлов

=== мультиплексирование: 50 клиентов на каждый endpoint ===
напрямую (без пулера)          server-бэкендов при 50 клиентах: 50  (всего pgbench-бэкендов в базе: 50)
через PgBouncer (pool=5)        server-бэкендов при 50 клиентах: 5   (всего pgbench-бэкендов в базе: 5)
через pgcat (pool=5)            server-бэкендов при 50 клиентах: 5   (всего pgbench-бэкендов в базе: 10)
через Odyssey (pool=5)          server-бэкендов при 50 клиентах: 5   (всего pgbench-бэкендов в базе: 15)

# Напрямую: 50 клиентов = 50 процессов на сервере (по процессу на соединение).
# Через любой пулер: 50 клиентов обслуживаются горсткой реальных соединений.
# ВСЕ ТРИ пулера при pool_size=5 держат РОВНО 5 server-соединений.
#
# ⚠️ ОТОЗВАНО (прежняя редакция): «pgcat и Odyssey держат чуть больше — 10 и 15,
# у каждого своя политика роста/резерва пула». Это был артефакт измерения, а не
# поведение пулеров. Счёт шёл по всем pgbench-бэкендам базы, без привязки к тому,
# ЧЕЙ это пул, а пулер после ухода клиентов держит свои server-соединения в пуле
# (idle до server_idle_timeout). Каждый следующий замер видел свои 5 плюс остаток
# от предыдущих — отсюда ряд 5/10/15. Колонка «всего» выше воспроизводит его в
# точности: это сумма, а не поведение.
#
# Проверка: при ОБРАТНОМ порядке (Odyssey -> pgcat -> PgBouncer) «своих» снова по 5
# у каждого — то есть прежний ряд переезжал вместе с порядком запуска.
#
# Теперь замер привязан к client_addr конкретного пулера, а сам пулер
# перезапускается перед своим замером (пул стартует пустым).
