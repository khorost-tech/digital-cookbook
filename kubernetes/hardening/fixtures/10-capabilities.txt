## РАЗБОР МАСКИ CapEff = 00000000a80425fb (разбор по документации, НЕ вывод команды)
#   Биты в маске сопоставлены с именами capabilities по порядку из ядерного
#   include/uapi/linux/capability.h. 0xa80425fb = 14 установленных бит:
#     bit  0  CAP_CHOWN            — менять владельца любых файлов
#     bit  1  CAP_DAC_OVERRIDE     — обходить проверки прав чтения/записи/выполнения
#     bit  3  CAP_FOWNER           — действовать как владелец файла (chmod, utime и т.п.)
#     bit  4  CAP_FSETID           — сохранять setuid/setgid при изменении файла
#     bit  5  CAP_KILL             — слать сигналы любым процессам
#     bit  6  CAP_SETGID           — менять gid процесса
#     bit  7  CAP_SETUID           — менять uid процесса (ключ к смене личности)
#     bit  8  CAP_SETPCAP          — менять capabilities других процессов
#     bit 10  CAP_NET_BIND_SERVICE — слушать привилегированные порты (<1024)
#     bit 13  CAP_NET_RAW          — RAW/PACKET-сокеты (ping, спуфинг, сниффинг)
#     bit 18  CAP_SYS_CHROOT       — chroot()
#     bit 27  CAP_MKNOD            — создавать спец-файлы устройств
#     bit 29  CAP_AUDIT_WRITE      — писать в аудит-лог ядра
#     bit 31  CAP_SETFCAP          — назначать capabilities файлам
#   ИТОГО 14 capabilities. Ключевые для атак: CAP_SETUID/CAP_SETGID (смена личности),
#   CAP_DAC_OVERRIDE/CAP_CHOWN/CAP_FOWNER (обход прав ФС), CAP_NET_RAW (сеть, спуфинг).
#   (Часто цитируемое «15 capabilities» — из памяти; фактический popcount маски a80425fb = 14.)

## ДЕМО 1в — ЧТО ЛОМАЕТ readOnlyRootFilesystem И КАК ЧИНИТСЯ
#   readOnlyRootFilesystem не «ломает приложение» — он требует явно объявить, куда оно пишет.

# cmd: kubectl -n cookbook-hardening exec hardened-pod -- sh -c 'echo data > /tmp/cache.dat'
sh: can't create /tmp/cache.dat: Read-only file system
command terminated with exit code 1
# Приложению, пишущему во временный каталог (/tmp — кэш, локи, сессии), read-only корень мешает.

# cmd: kubectl -n cookbook-hardening exec hardened-writable -- sh -c 'echo data > /tmp/cache.dat && echo OK && cat /tmp/cache.dat'
#   hardened-writable = тот же restricted-под + emptyDir, смонтированный в /tmp.
OK
data
# Штатное решение: смонтировать emptyDir в нужный путь — запись в /tmp проходит.

# cmd: kubectl -n cookbook-hardening exec hardened-writable -- sh -c 'touch /rootfile && echo "запись в / разрешена" || echo "запись в / по-прежнему запрещена (Read-only)"'
touch: /rootfile: Read-only file system
запись в / по-прежнему запрещена (Read-only)
# При этом сам корень остаётся read-only — открыт ровно один явно объявленный путь, не вся ФС.
